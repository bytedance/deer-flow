import json
with open('/mnt/user-data/outputs/charts.json') as f:
    d = json.load(f)

# map chart_type to component
type_map = {'card': 'card', 'trend': 'echart', 'waveform': 'echart', 'table': 'table'}
for b in d['blocks']:
    b['component'] = type_map.get(b['chart_type'], b['component'])
    # remove chart_type key
    b.pop('chart_type', None)

with open('/mnt/user-data/outputs/charts.json', 'w') as f:
